        // === Firebase 初始化與認證 ===

        async function initializeFirebase() {

            try {

                const app = initializeApp(firebaseConfig);

                db = getFirestore(app);

                auth = getAuth(app);

                

                if (initialAuthToken) {

                    await signInWithCustomToken(auth, initialAuthToken);

                } else {

                    await signInAnonymously(auth);

                }


                onAuthStateChanged(auth, (user) => {

                    if (user) {

                        userId = user.uid;

                        authStatusElement.innerHTML = `<i class="ph-bold ph-user-circle"></i> <span>用戶 ID: ${userId.substring(0, 8)}...</span>`;

                        startListeningToData();

                    } else {

                        userId = null;

                        authStatusElement.innerHTML = `<i class="ph-bold ph-warning"></i> <span>未認證用戶</span>`;

                    }

                });


            } catch (error) {

                authStatusElement.innerHTML = `<i class="ph-bold ph-x-circle text-red-400"></i> <span>初始化錯誤</span>`;

            }

        }


        // === Firestore 數據監聽與處理 ===

        function startListeningToData() {

            if (!db || !userId) return;


            const docPath = `/artifacts/${appId}/users/${userId}/itinerary/${TRIP_DOC_ID}`;

            const tripDocRef = doc(db, docPath);


            onSnapshot(tripDocRef, (docSnapshot) => {

                if (docSnapshot.exists()) {

                    itineraryData = docSnapshot.data();

                    console.log("Data loaded from Firestore:", itineraryData);

                    renderAll();

                } else {

                    // 如果文件不存在，寫入初始數據

                    console.log("No data found. Initializing trip data...");

                    itineraryData = initialItinerary;

                    setDoc(tripDocRef, itineraryData).catch(e => console.error("Error writing initial data:", e));

                }

                loadingMessage.classList.add('hidden');

            }, (error) => {

                console.error("Firestore Snapshot Error:", error);

                loadingMessage.innerHTML = `<p class="text-red-500">資料載入失敗: ${error.message}</p>`;

            });

        }

        async function updateData() {

            if (!db || !userId) return;

            const docPath = `/artifacts/${appId}/users/${userId}/itinerary/${TRIP_DOC_ID}`;

            const tripDocRef = doc(db, docPath);

            // 使用 setTimeout 實現 debounce，避免過度寫入

            await setDoc(tripDocRef, itineraryData);

        }


        window.deleteItem = async function(dayKey, itemId) {

            if (!confirm('確認刪除此行程項目嗎？')) return;


            const dayItems = itineraryData[dayKey];

            if (!dayItems) return;


            itineraryData[dayKey] = dayItems.filter(item => item.id !== itemId);

            await updateData();

        }
